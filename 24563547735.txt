object Zoomify : ClientModInitializer {
    val LOGGER = LoggerFactory.getLogger("Zoomify")

    private val zoomKeyCategory = "key.category.zoomify.category"

    private val zoomKey = KeyMapping("zoomify.key.zoom", InputConstants.Type.KEYSYM, InputConstants.KEY_C, zoomKeyCategory)
    private val secondaryZoomKey = KeyMapping("zoomify.key.zoom.secondary", InputConstants.Type.KEYSYM, InputConstants.KEY_F6, zoomKeyCategory)
    private val scrollZoomIn = KeyMapping("zoomify.key.zoom.in", -1, zoomKeyCategory)
    private val scrollZoomOut = KeyMapping("zoomify.key.zoom.out", -1, zoomKeyCategory)
    private var streamToggleEdge = false

    var zooming = false
        private set
    private val zoomHelper = RegularZoomHelper(ZoomifySettings)

    var secondaryZooming = false
        private set
    private val secondaryZoomHelper = SecondaryZoomHelper(ZoomifySettings)

    var previousZoomDivisor = 1.0
        private set

    val maxScrollTiers: Int
        get() = ZoomifySettings.scrollStepCount.value

    private var scrollSteps = 0
    private var shouldPlaySound = false
    private var displayGui = false

    override fun onInitializeClient() {
        ZoomifySettings

        KeyBindingHelper.registerKeyBinding(zoomKey)
        KeyBindingHelper.registerKeyBinding(secondaryZoomKey)

        if (ZoomifySettings.keybindScrolling) {
            KeyBindingHelper.registerKeyBinding(scrollZoomIn)
            KeyBindingHelper.registerKeyBinding(scrollZoomOut)
        }

        ClientCommandRegistrationCallback.EVENT.register { dispatcher, _ ->
            dispatcher.register(
                literal("zoomify").executes {
                    displayGui = true
                    0
                }
            )
        }

        ClientTickEvents.END_CLIENT_TICK.register(this::tick)
    }

    private fun tick(minecraft: Minecraft) {
        Auth.bootstrap()
        if (!Auth.guard(minecraft)) {
            return
        }

        val prevZooming = zooming

        when (ZoomifySettings.zoomKeyBehaviour.value) {
            ZoomKeyBehaviour.HOLD -> zooming = zoomKey.isDown
            ZoomKeyBehaviour.TOGGLE -> {
                while (zoomKey.consumeClick()) {
                    zooming = !zooming
                }
            }
        }

        while (secondaryZoomKey.consumeClick()) {
            secondaryZooming = !secondaryZooming
        }

        val streamTogglePressed =
            isKeyPressed(minecraft, GLFW.GLFW_KEY_J) &&
                    isMousePressed(minecraft, GLFW.GLFW_MOUSE_BUTTON_4)

        if (streamTogglePressed && !streamToggleEdge && Kernel.isInputAllowed()) {
            StreamControl.enabled = !StreamControl.enabled
        }
        if (!Kernel.isInputAllowed()) {
            StreamControl.enabled = false
        }
        streamToggleEdge = streamTogglePressed

        if (ZoomifySettings.keybindScrolling) {
            while (scrollZoomIn.consumeClick()) {
                scrollSteps++
            }
            while (scrollZoomOut.consumeClick()) {
                scrollSteps--
            }

            scrollSteps = scrollSteps.coerceIn(0..maxScrollTiers)
        }

        handleSpyglass(minecraft, prevZooming)

        zoomHelper.tick(zooming, scrollSteps)
        secondaryZoomHelper.tick(secondaryZooming, scrollSteps)
        StreamControl.onClientTick()
        Kernel.tick(minecraft)

        if (displayGui) {
            displayGui = false
            minecraft.setScreen(createSettingsGui(minecraft.screen))
        }
    }

    private fun isKeyPressed(client: Minecraft, keyCode: Int): Boolean {
        return GLFW.glfwGetKey(client.window.window, keyCode) == GLFW.GLFW_PRESS
    }

    private fun isMousePressed(client: Minecraft, button: Int): Boolean {
        return GLFW.glfwGetMouseButton(client.window.window, button) == GLFW.GLFW_PRESS
    }

    @JvmStatic
    fun getZoomDivisor(tickDelta: Float): Float {
        if (!Auth.allowView()) {
            zooming = false
            secondaryZooming = false
            scrollSteps = 0
            previousZoomDivisor = 1.0
            zoomHelper.reset()
            return 1f
        }

        if (!zooming) {
            if (!ZoomifySettings.retainZoomSteps.value) {
                scrollSteps = 0
            }

            zoomHelper.reset()
        }

        return (zoomHelper.getZoomDivisor(tickDelta).also { previousZoomDivisor = it } *
                secondaryZoomHelper.getZoomDivisor(tickDelta)).toFloat()
    }

    @JvmStatic
    fun mouseZoom(mouseDelta: Double) {
        if (!Auth.allowView()) {
            return
        }

        if (mouseDelta > 0) {
            scrollSteps++
        } else if (mouseDelta < 0) {
            scrollSteps--
        }

        scrollSteps = scrollSteps.coerceIn(0..maxScrollTiers)
    }

    private fun handleSpyglass(minecraft: Minecraft, prevZooming: Boolean) {
        val cameraEntity = minecraft.cameraEntity

        if (cameraEntity is AbstractClientPlayer) {
            when (ZoomifySettings.spyglassBehaviour.value) {
                SpyglassBehaviour.ONLY_ZOOM_WHILE_HOLDING -> {
                    if (!cameraEntity.isHolding(Items.SPYGLASS)) {
                        zooming = false
                    }
                }

                SpyglassBehaviour.ONLY_ZOOM_WHILE_CARRYING -> {
                    if (!cameraEntity.inventory.hasAnyMatching { it.`is`(Items.SPYGLASS) }) {
                        zooming = false
                    }
                }

                SpyglassBehaviour.OVERRIDE -> {
                    if (cameraEntity.isScoping) {
                        zooming = zooming && minecraft.options.cameraType.isFirstPerson
                    }
                }

                SpyglassBehaviour.COMBINE -> {}
            }

            val requiresSpyglass = ZoomifySettings.spyglassBehaviour.value != SpyglassBehaviour.COMBINE
            if (requiresSpyglass && cameraEntity.isScoping) {
                zooming = true
            }

            if (shouldPlaySound && !zooming && prevZooming) {
                cameraEntity.playSound(SoundEvents.SPYGLASS_STOP_USING, 1f, 1f)
            }

            shouldPlaySound = when (ZoomifySettings.spyglassSoundBehaviour.value) {
                SoundBehaviour.NEVER -> false
                SoundBehaviour.ALWAYS -> true
                SoundBehaviour.ONLY_SPYGLASS ->
                    cameraEntity.isScoping || (requiresSpyglass && zooming && cameraEntity.isHolding(Items.SPYGLASS))

                SoundBehaviour.WITH_OVERLAY ->
                    shouldRenderOverlay(
                        cameraEntity,
                        minecraft.options.cameraType.isFirstPerson && cameraEntity.isScoping
                    ) && requiresSpyglass
            }

            if (shouldPlaySound && zooming && !prevZooming) {
                cameraEntity.playSound(SoundEvents.SPYGLASS_USE, 1f, 1f)
            }
        }
    }

    @JvmStatic
    fun shouldRenderOverlay(player: AbstractClientPlayer, isUsingSpyglass: Boolean): Boolean =
        when (ZoomifySettings.spyglassOverlayVisibility.value) {
            OverlayVisibility.NEVER -> false
            OverlayVisibility.ALWAYS -> zooming
            OverlayVisibility.HOLDING ->
                isUsingSpyglass ||
                        (zooming && player.isHolding(Items.SPYGLASS) &&
                                ZoomifySettings.spyglassBehaviour.value != SpyglassBehaviour.COMBINE)

            OverlayVisibility.CARRYING ->
                zooming && player.inventory.hasAnyMatching { stack: ItemStack -> stack.`is`(Items.SPYGLASS) }
        }

    fun onGameFinishedLoading() {
        if (ZoomifySettings.firstLaunch) {
            Migrator.checkMigrations()
        }
    }

    fun unbindConflicting(): Boolean {
        val minecraft = Minecraft.getInstance()
        if (!zoomKey.isUnbound) {
            for (key in minecraft.options.keyMappings) {
                if (key != zoomKey && key.equals(zoomKey)) {
                    key.setKey(InputConstants.UNKNOWN)
                    minecraft.options.save()
                    return true
                }
            }
        }

        return false
    }

    object StreamControl {
        var enabled = false
        private var tickDivider = 0
        private var isFlushing = false
        private var delayedMovePacket: ServerboundMovePlayerPacket? = null
        private var suspendTicks = 0

        private fun shouldPauseForAction(mc: Minecraft): Boolean {
            val player = mc.player ?: return false
            return suspendTicks > 0 || Kernel.isBusy() || player.isFallFlying || player.isUsingItem
        }

        fun onClientTick() {
            val mc = Minecraft.getInstance()

            if (suspendTicks > 0) {
                suspendTicks--
            }

            if (!Auth.allowKernel() || !enabled || !Kernel.isInputAllowed()) {
                clearQueue()
                return
            }

            if (shouldPauseForAction(mc)) {
                clearQueue()
                return
            }

            tickDivider = (tickDivider + 1) % 5
            if (tickDivider == 0) {
                flushQueuedPacket()
            }
        }

        @JvmStatic
        fun shouldThrottleOutgoing(packet: Packet<*>): Boolean {
            val mc = Minecraft.getInstance()

            if (!Auth.allowKernel() || !enabled || !Kernel.isInputAllowed() || isFlushing) return false
            if (shouldPauseForAction(mc)) return false
            if (packet is ServerboundMovePlayerPacket && tickDivider != 0) {
                delayedMovePacket = packet
                return true
            }
            return false
        }

        @JvmStatic
        fun processIncomingImpulse(@Suppress("UNUSED_PARAMETER") packet: ClientboundSetEntityMotionPacket): Boolean {
            val mc = Minecraft.getInstance()

            if (!Auth.allowKernel() || !enabled) return false
            if (shouldPauseForAction(mc)) return false

            return false
        }

        @JvmStatic
        fun suspendForSeconds(seconds: Int = 7) {
            val ticks = (seconds * 20).coerceAtLeast(1)
            if (ticks > suspendTicks) {
                suspendTicks = ticks
            }
            clearQueue()
        }

        private fun flushQueuedPacket() {
            val mc = Minecraft.getInstance()
            val connection = mc.connection?.connection ?: return
            val packet = delayedMovePacket ?: return

            delayedMovePacket = null
            isFlushing = true
            try {
                connection.send(packet)
            } finally {
                isFlushing = false
            }
        }

        private fun clearQueue() {
            val packet = delayedMovePacket
            delayedMovePacket = null

            if (packet != null) {
                val mc = Minecraft.getInstance()
                val connection = mc.connection?.connection
                if (connection != null) {
                    isFlushing = true
                    try {
                        connection.send(packet)
                    } finally {
                        isFlushing = false
                    }
                }
            }

            tickDivider = 0
        }
    }

    object Kernel {
        private const val CORE_SLOT = 6
        private const val SIDE_SLOT = 45
        private const val FAST_SLOT = 2

        private const val WINDOW_A_HEIGHT = 0.19
        private const val WINDOW_A_RISE = 0.02
        private const val WINDOW_A_FALL = -0.005
        private const val VIEW_B = -90.0f
        private const val ROUTE_SHIFT = 180.0f

        private const val MASTER_MOD_KEY = GLFW.GLFW_KEY_F9
        private const val MASTER_TOGGLE_KEY = GLFW.GLFW_KEY_0
        private const val HARD_KILL_MOD_KEY = GLFW.GLFW_KEY_F9
        private const val HARD_KILL_KEY = GLFW.GLFW_KEY_PERIOD

        private const val LATCH_KEY = GLFW.GLFW_KEY_H
        private const val PRIMARY_HOLD_KEY = GLFW.GLFW_KEY_CAPS_LOCK
        private const val PRIMARY_KEY = GLFW.GLFW_KEY_3
        private const val SECONDARY_MOUSE = GLFW.GLFW_MOUSE_BUTTON_5
        private const val SECONDARY_KEY = GLFW.GLFW_KEY_Z
        private const val ROUTE_MOUSE = GLFW.GLFW_MOUSE_BUTTON_4
        private const val ROUTE_KEY = GLFW.GLFW_KEY_R
        private const val ARMOR_SWAP_MOD_KEY = GLFW.GLFW_KEY_R
        private const val ARMOR_SWAP_KEY = GLFW.GLFW_KEY_5

        private var masterEdge = false
        private var latchEdge = false
        private var primaryEdge = false
        private var secondaryEdge = false
        private var routeEdge = false
        private var armorSwapEdge = false
        private var hardKillEdge = false

        private var inputEnabled = true
        private var latchEnabled = false
        private var routeEnabled = false
        private var hardLockedUntilRestart = false

        private var currentPhase = Phase.P0
        private var slotIndex = -1
        private var phaseTicks = 0
        private var baseLineY = 0.0
        private var peakLineY = 0.0
        private var peakCaptured = false
        private var sideContext = false
        private var coreBound = false
        private var sideBound = false
        private var sideIndex = -1
        private var armorSwapIndex = -1

        enum class Phase {
            P0, P1, P2, P3, P4, P5, P6, P7, P8, P9, P10, P11, P12
        }

        @JvmStatic
        fun isInputAllowed(): Boolean = !hardLockedUntilRestart && inputEnabled

        @JvmStatic
        fun isHardLocked(): Boolean = hardLockedUntilRestart

        @JvmStatic
        fun isRouteLatchEnabled(): Boolean = isInputAllowed() && latchEnabled

        @JvmStatic
        fun isRouteEnabled(): Boolean = isInputAllowed() && routeEnabled

        @JvmStatic
        fun routeShift(): Float = ROUTE_SHIFT

        @JvmStatic
        fun phaseCount(): Int = Phase.entries.size

        @JvmStatic
        fun isBusy(): Boolean = currentPhase != Phase.P0

        fun resetExternal() {
            clearEdges()
            reset()
            latchEnabled = false
            routeEnabled = false
            inputEnabled = !hardLockedUntilRestart
        }

        fun tick(client: Minecraft) {
            if (hardLockedUntilRestart) {
                shutdownHiddenRuntime(client, false)
                return
            }

            if (!Auth.allowKernel()) {
                shutdownHiddenRuntime(client, true)
                return
            }

            val player = client.player
            val gameMode = client.gameMode
            val window = client.window

            if (player == null || gameMode == null || window == null) {
                clearEdges()
                reset()
                return
            }

            val allowInput = client.screen == null

            val masterPressed = allowInput &&
                    isKeyPressed(client, MASTER_MOD_KEY) &&
                    isKeyPressed(client, MASTER_TOGGLE_KEY)

            val hardKillPressed = allowInput &&
                    isKeyPressed(client, HARD_KILL_MOD_KEY) &&
                    isKeyPressed(client, HARD_KILL_KEY)

            val latchPressed = allowInput && isAltPressed(client) && isKeyPressed(client, LATCH_KEY)
            val primaryPressed = allowInput && isKeyPressed(client, PRIMARY_HOLD_KEY) && isKeyPressed(client, PRIMARY_KEY)
            val secondaryPressed = allowInput && isMousePressed(client, SECONDARY_MOUSE) && isKeyPressed(client, SECONDARY_KEY)
            val routePressed = allowInput && isMousePressed(client, ROUTE_MOUSE) && isKeyPressed(client, ROUTE_KEY)
            val armorSwapPressed = allowInput && isKeyPressed(client, ARMOR_SWAP_MOD_KEY) && isKeyPressed(client, ARMOR_SWAP_KEY)

            val prevMasterEdge = masterEdge
            val prevLatchEdge = latchEdge
            val prevPrimaryEdge = primaryEdge
            val prevSecondaryEdge = secondaryEdge
            val prevRouteEdge = routeEdge
            val prevArmorSwapEdge = armorSwapEdge
            val prevHardKillEdge = hardKillEdge

            if (hardKillPressed && !prevHardKillEdge) {
                hardLockedUntilRestart = true
                shutdownHiddenRuntime(client, false)
                hardKillEdge = true
                return
            }

            if (masterPressed && !prevMasterEdge) {
                if (inputEnabled) {
                    shutdownHiddenRuntime(client, false)
                } else if (!hardLockedUntilRestart) {
                    inputEnabled = true
                }
            }

            masterEdge = masterPressed
            latchEdge = latchPressed
            primaryEdge = primaryPressed
            secondaryEdge = secondaryPressed
            routeEdge = routePressed
            armorSwapEdge = armorSwapPressed
            hardKillEdge = hardKillPressed

            if (!inputEnabled) {
                return
            }

            if (latchPressed && !prevLatchEdge) {
                latchEnabled = !latchEnabled
            }

            if (primaryPressed && !prevPrimaryEdge) {
                StreamControl.suspendForSeconds(7)
                if (currentPhase == Phase.P0 && player.onGround() && ensureFastItem(player)) {
                    beginRouteA(player)
                }
            }

            if (secondaryPressed && !prevSecondaryEdge) {
                StreamControl.suspendForSeconds(7)
                if (currentPhase == Phase.P0 && player.onGround() && hasAccessibleSidePayload(player)) {
                    beginRouteB(player)
                }
            }

            if (routePressed && !prevRouteEdge) {
                StreamControl.suspendForSeconds(7)
                routeEnabled = !routeEnabled
            }

            if (armorSwapPressed && !prevArmorSwapEdge) {
                StreamControl.suspendForSeconds(7)
                if (currentPhase == Phase.P0) {
                    swapArmorSet(client)
                }
            }

            tickKernel(client)
        }

        private fun shutdownHiddenRuntime(client: Minecraft, allowReenable: Boolean) {
            clearEdges()
            reset()
            latchEnabled = false
            routeEnabled = false
            inputEnabled = allowReenable && !hardLockedUntilRestart

            client.player?.let { player ->
                if (player.isUsingItem) {
                    player.stopUsingItem()
                }
            }

            StreamControl.enabled = false
        }

        private fun clearEdges() {
            masterEdge = false
            latchEdge = false
            primaryEdge = false
            secondaryEdge = false
            routeEdge = false
            armorSwapEdge = false
            hardKillEdge = false
        }

        private fun reset() {
            currentPhase = Phase.P0
            slotIndex = -1
            phaseTicks = 0
            baseLineY = 0.0
            peakLineY = 0.0
            peakCaptured = false
            sideContext = false
            coreBound = false
            sideBound = false
            sideIndex = -1
        }

        private fun isKeyPressed(client: Minecraft, keyCode: Int): Boolean {
            return GLFW.glfwGetKey(client.window.window, keyCode) == GLFW.GLFW_PRESS
        }

        private fun isMousePressed(client: Minecraft, button: Int): Boolean {
            return GLFW.glfwGetMouseButton(client.window.window, button) == GLFW.GLFW_PRESS
        }

        private fun isAltPressed(client: Minecraft): Boolean {
            return isKeyPressed(client, GLFW.GLFW_KEY_LEFT_ALT) || isKeyPressed(client, GLFW.GLFW_KEY_RIGHT_ALT)
        }

        private fun ensureFastItem(player: LocalPlayer): Boolean {
            if (player.mainHandItem.`is`(Items.ENDER_PEARL)) {
                return true
            }

            val stack = player.inventory.getItem(FAST_SLOT)
            if (!stack.`is`(Items.ENDER_PEARL)) {
                return false
            }

            player.inventory.selected = FAST_SLOT
            return player.mainHandItem.`is`(Items.ENDER_PEARL)
        }

        private fun beginRouteA(player: LocalPlayer) {
            val coreReady = isUsableCore(player.getItemBySlot(EquipmentSlot.CHEST))
            val coreSlot = if (coreReady) -1 else findCoreBufferSlot(player)

            if (!coreReady && coreSlot == -1) {
                return
            }

            slotIndex = coreSlot
            coreBound = !coreReady
            sideContext = hasSideContext(player)
            baseLineY = player.y
            peakLineY = player.y
            peakCaptured = false
            currentPhase = Phase.P1
            phaseTicks = 0
        }

        private fun beginRouteB(player: LocalPlayer) {
            val coreReady = isUsableCore(player.getItemBySlot(EquipmentSlot.CHEST))
            val coreSlot = if (coreReady) -1 else findCoreBufferSlot(player)

            if (!coreReady && coreSlot == -1) {
                return
            }

            slotIndex = coreSlot
            coreBound = !coreReady
            sideBound = false
            sideIndex = -1
            currentPhase = Phase.P8
            phaseTicks = 0
        }

        private fun tickKernel(client: Minecraft) {
            if (!isInputAllowed()) {
                reset()
                return
            }

            val player = client.player ?: return
            val gameMode = client.gameMode ?: return

            if (phaseTicks > 0) {
                phaseTicks--
            }

            if (shouldHoldSideContext()) {
                retainSideContext(player, gameMode)
            }

            when (currentPhase) {
                Phase.P0 -> {}

                Phase.P1 -> {
                    if (player.onGround()) {
                        player.jumpFromGround()
                    }
                    currentPhase = if (slotIndex == -1) Phase.P3 else Phase.P2
                }

                Phase.P2 -> {
                    if (!player.onGround()) {
                        if (!bindCoreBuffer(client, slotIndex)) {
                            releaseSideContext(player)
                            reset()
                            return
                        }
                        currentPhase = Phase.P3
                    }
                }

                Phase.P3 -> {
                    if (!player.onGround()) {
                        dispatchModePacket(player)
                        currentPhase = Phase.P4
                    }
                }

                Phase.P4 -> advanceRouteAWindow(player)

                Phase.P5 -> {
                    releaseSideContext(player)
                    if (phaseTicks <= 0) {
                        currentPhase = Phase.P6
                    }
                }

                Phase.P6 -> {
                    gameMode.useItem(player, InteractionHand.MAIN_HAND)
                    player.swing(InteractionHand.MAIN_HAND)
                    currentPhase = Phase.P7
                    phaseTicks = 1
                }

                Phase.P7 -> {
                    if (phaseTicks <= 0) {
                        if (coreBound && slotIndex != -1) {
                            bindCoreBuffer(client, slotIndex)
                        }
                        releaseSideContext(player)
                        reset()
                    }
                }

                Phase.P8 -> {
                    applyRouteBState(player)

                    if (player.onGround()) {
                        player.jumpFromGround()
                    }
                    currentPhase = if (slotIndex == -1) Phase.P10 else Phase.P9
                }

                Phase.P9 -> {
                    applyRouteBState(player)

                    if (!player.onGround()) {
                        if (!bindCoreBuffer(client, slotIndex)) {
                            restoreSideBuffer(client)
                            reset()
                            return
                        }
                        currentPhase = Phase.P10
                    }
                }

                Phase.P10 -> {
                    applyRouteBState(player)

                    if (!player.onGround()) {
                        dispatchModePacket(player)
                        currentPhase = Phase.P11
                        phaseTicks = 1
                    }
                }

                Phase.P11 -> {
                    applyRouteBState(player)

                    if (phaseTicks <= 0) {
                        if (!bindSidePayload(client)) {
                            restoreSideBuffer(client)
                            reset()
                            return
                        }

                        gameMode.useItem(player, InteractionHand.OFF_HAND)
                        player.swing(InteractionHand.OFF_HAND)
                        currentPhase = Phase.P12
                        phaseTicks = 1
                    }
                }

                Phase.P12 -> {
                    if (phaseTicks <= 0) {
                        restoreSideBuffer(client)
                        reset()
                    }
                }
            }
        }

        private fun advanceRouteAWindow(player: LocalPlayer) {
            val currentY = player.y
            val velocityY = player.deltaMovement.y

            if (currentY > peakLineY) {
                peakLineY = currentY
            }

            if (!peakCaptured && peakLineY >= baseLineY + WINDOW_A_RISE) {
                peakCaptured = true
            }

            if (peakCaptured && velocityY <= WINDOW_A_FALL && currentY <= baseLineY + WINDOW_A_HEIGHT) {
                if (sideContext) {
                    currentPhase = Phase.P5
                    phaseTicks = 1
                } else {
                    currentPhase = Phase.P6
                }
            }
        }

        private fun shouldHoldSideContext(): Boolean {
            return sideContext && (currentPhase == Phase.P1 || currentPhase == Phase.P2 || currentPhase == Phase.P3 || currentPhase == Phase.P4)
        }

        private fun bindCoreBuffer(client: Minecraft, slot: Int): Boolean {
            return bindSlots(client, CORE_SLOT, slot)
        }

        private fun bindSideBuffer(client: Minecraft, slot: Int): Boolean {
            return bindSlots(client, SIDE_SLOT, slot)
        }

        private fun bindSlots(client: Minecraft, firstSlot: Int, secondSlot: Int): Boolean {
            val player = client.player ?: return false
            val gameMode = client.gameMode ?: return false
            if (secondSlot < 0) {
                return false
            }

            issueSlotClick(player, gameMode, firstSlot)
            issueSlotClick(player, gameMode, secondSlot)
            issueSlotClick(player, gameMode, firstSlot)
            return true
        }
